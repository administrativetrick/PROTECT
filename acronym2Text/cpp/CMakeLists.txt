cmake_minimum_required(VERSION 3.10)
project(acronym2text CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_executable(acronym2text acronym2text.cpp)
if(NOT MSVC)
  target_compile_options(acronym2text PRIVATE -Wall -Wextra -O2)
endif()

find_package(Qt5 COMPONENTS Widgets QUIET)
find_package(PkgConfig QUIET)
if(Qt5_FOUND AND PkgConfig_FOUND)
  pkg_check_modules(X11HOOK QUIET x11 xtst)
  if(X11HOOK_FOUND)
    set(CMAKE_AUTOMOC ON)
    add_executable(acronym2text_gui acronym2text_gui.cpp)
    target_include_directories(acronym2text_gui PRIVATE ${X11HOOK_INCLUDE_DIRS})
    target_link_libraries(acronym2text_gui PRIVATE Qt5::Widgets ${X11HOOK_LIBRARIES})
    if(NOT MSVC)
      target_compile_options(acronym2text_gui PRIVATE -Wall -Wextra -O2)
    endif()
  else()
    message(STATUS "X11/Xtst not found; skipping acronym2text_gui")
  endif()
else()
  message(STATUS "Qt5/PkgConfig not found; skipping acronym2text_gui")
endif()
