cmake_minimum_required(VERSION 3.10)
project(imgmetadatascraper CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(SQLite3 REQUIRED)
add_executable(imgmetadatascraper imgmetadatascraper.cpp)
target_link_libraries(imgmetadatascraper PRIVATE SQLite::SQLite3)
if(NOT MSVC)
  target_compile_options(imgmetadatascraper PRIVATE -Wall -Wextra -O2)
endif()
