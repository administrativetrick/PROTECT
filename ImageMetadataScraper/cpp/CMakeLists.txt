cmake_minimum_required(VERSION 3.10)
project(imgmetadatascraper CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(SQLite3 REQUIRED)

add_executable(imgmetadatascraper imgmetadatascraper.cpp)
target_link_libraries(imgmetadatascraper PRIVATE SQLite::SQLite3)
if(NOT MSVC)
  target_compile_options(imgmetadatascraper PRIVATE -Wall -Wextra -O2)
endif()

find_package(Qt5 COMPONENTS Widgets QUIET)
if(Qt5_FOUND)
  set(CMAKE_AUTOMOC ON)
  add_executable(imgmetadatascraper_gui imgmetadatascraper_gui.cpp)
  target_link_libraries(imgmetadatascraper_gui PRIVATE Qt5::Widgets SQLite::SQLite3)
  if(NOT MSVC)
    target_compile_options(imgmetadatascraper_gui PRIVATE -Wall -Wextra -O2)
  endif()
else()
  message(STATUS "Qt5 not found; skipping imgmetadatascraper_gui")
endif()
