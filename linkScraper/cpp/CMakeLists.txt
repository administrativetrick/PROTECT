cmake_minimum_required(VERSION 3.10)
project(linkscraper CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(SQLite3 REQUIRED)

add_executable(linkscraper linkscraper.cpp)
target_link_libraries(linkscraper PRIVATE SQLite::SQLite3)
if(NOT MSVC)
  target_compile_options(linkscraper PRIVATE -Wall -Wextra -O2)
endif()

find_package(Qt5 COMPONENTS Widgets QUIET)
if(Qt5_FOUND)
  set(CMAKE_AUTOMOC ON)
  add_executable(linkscraper_gui linkscraper_gui.cpp)
  target_link_libraries(linkscraper_gui PRIVATE Qt5::Widgets SQLite::SQLite3)
  if(NOT MSVC)
    target_compile_options(linkscraper_gui PRIVATE -Wall -Wextra -O2)
  endif()
else()
  message(STATUS "Qt5 not found; skipping linkscraper_gui")
endif()
