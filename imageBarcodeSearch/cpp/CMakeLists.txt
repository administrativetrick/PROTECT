cmake_minimum_required(VERSION 3.10)
project(imagebarcodesearch CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(SQLite3 REQUIRED)
find_package(PNG REQUIRED)
find_package(JPEG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(ZBAR REQUIRED zbar)

add_executable(imagebarcodesearch imagebarcodesearch.cpp)
target_include_directories(imagebarcodesearch PRIVATE ${ZBAR_INCLUDE_DIRS})
target_link_libraries(imagebarcodesearch PRIVATE SQLite::SQLite3 PNG::PNG JPEG::JPEG ${ZBAR_LIBRARIES})
if(NOT MSVC)
  target_compile_options(imagebarcodesearch PRIVATE -Wall -Wextra -O2)
endif()

find_package(Qt5 COMPONENTS Widgets QUIET)
if(Qt5_FOUND)
  set(CMAKE_AUTOMOC ON)
  add_executable(imagebarcodesearch_gui imagebarcodesearch_gui.cpp)
  target_include_directories(imagebarcodesearch_gui PRIVATE ${ZBAR_INCLUDE_DIRS})
  target_link_libraries(imagebarcodesearch_gui PRIVATE Qt5::Widgets SQLite::SQLite3 PNG::PNG JPEG::JPEG ${ZBAR_LIBRARIES})
  if(NOT MSVC)
    target_compile_options(imagebarcodesearch_gui PRIVATE -Wall -Wextra -O2)
  endif()
else()
  message(STATUS "Qt5 not found; skipping imagebarcodesearch_gui")
endif()
