cmake_minimum_required(VERSION 3.10)
project(websitecloner CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(SQLite3 REQUIRED)

add_executable(websitecloner websitecloner.cpp)
target_link_libraries(websitecloner PRIVATE SQLite::SQLite3)
if(NOT MSVC)
  target_compile_options(websitecloner PRIVATE -Wall -Wextra -O2)
endif()

find_package(Qt5 COMPONENTS Widgets QUIET)
if(Qt5_FOUND)
  set(CMAKE_AUTOMOC ON)
  add_executable(websitecloner_gui websitecloner_gui.cpp)
  target_link_libraries(websitecloner_gui PRIVATE Qt5::Widgets SQLite::SQLite3)
  if(NOT MSVC)
    target_compile_options(websitecloner_gui PRIVATE -Wall -Wextra -O2)
  endif()
else()
  message(STATUS "Qt5 not found; skipping websitecloner_gui")
endif()
